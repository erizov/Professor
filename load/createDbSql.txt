CREATE TYPE edu_level AS ENUM ('school', 'university');
CREATE TYPE lang_code AS ENUM ('en', 'ru');

CREATE TABLE algorithms (
  algorithm_name TEXT PRIMARY KEY,
  canonical_label TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

CREATE TABLE algorithm_descriptions (
  id SERIAL PRIMARY KEY,
  algorithm_name TEXT NOT NULL REFERENCES algorithms(algorithm_name) ON DELETE CASCADE,
  language lang_code NOT NULL,
  level edu_level NOT NULL,
  title TEXT,
  short_description TEXT,
  long_description TEXT,
  example_snippet TEXT,
  source_url TEXT,
  source_site TEXT,
  fetched_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
  quality_score REAL DEFAULT 0.0,
  UNIQUE (algorithm_name, language, level)
);

CREATE TABLE algorithm_aliases (
  id SERIAL PRIMARY KEY,
  algorithm_name TEXT NOT NULL REFERENCES algorithms(algorithm_name) ON DELETE CASCADE,
  alias TEXT NOT NULL,
  language lang_code,
  source_url TEXT
);

CREATE TABLE fetch_logs (
  id SERIAL PRIMARY KEY,
  algorithm_name TEXT,
  language lang_code,
  level edu_level,
  status TEXT,
  message TEXT,
  fetched_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);
